cmake_minimum_required(VERSION 3.25)
project(Chopperator VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
foreach(target IN ITEMS chopper_core_tests chopper_extended_tests)
  if(target STREQUAL "chopper_core_tests")
    add_executable(${target} tests/CoreTests.cpp)
  else()
    add_executable(${target} tests/ExtendedTests.cpp)
  endif()
  target_include_directories(${target} PRIVATE src)
  target_compile_features(${target} PRIVATE cxx_std_23)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME core_contract COMMAND chopper_core_tests)
add_test(NAME extended_contract COMMAND chopper_extended_tests)
add_test(NAME release_negative_control COMMAND chopper_core_tests --negative)
set_tests_properties(release_negative_control PROPERTIES WILL_FAIL TRUE)
