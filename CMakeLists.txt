cmake_minimum_required(VERSION 3.25)
project(Chopperator VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_executable(chopper_core_tests tests/CoreTests.cpp)
target_include_directories(chopper_core_tests PRIVATE src)
target_compile_features(chopper_core_tests PRIVATE cxx_std_23)
if(MSVC)
  target_compile_options(chopper_core_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(chopper_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME core_contract COMMAND chopper_core_tests)
add_test(NAME release_negative_control COMMAND chopper_core_tests --negative)
set_tests_properties(release_negative_control PROPERTIES WILL_FAIL TRUE)
